# 全体テスト（ログあり）
PROJECT="/Users/naokiyamada/git/guitar-mes/guitar_mes"
DAY=$(date '+%Y%m%d')
RUN_ID=$(date '+%Y%m%d-%H%M%S')
LOGDIR="$PROJECT/logs/test-logs/$DAY"
LOGFILE="$LOGDIR/console-$RUN_ID.log"
mkdir -p "$LOGDIR"
(
  set -o pipefail
  "$PROJECT/mvnw" \
    -f "$PROJECT/pom.xml" \
    -Dstyle.color=always \
    test \
    2>&1 | tee "$LOGFILE"
)
STATUS=$?
echo
echo "終了コード: $STATUS"
echo "ログ: $LOGFILE"
if [ "$STATUS" -ne 0 ]; then
  echo "通常テスト全件に失敗しました。"
else
  echo "通常テスト全件は正常に完了しました。"
fi
