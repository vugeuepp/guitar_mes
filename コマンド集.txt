{
  PROJECT="/Users/naokiyamada/git/guitar-mes/guitar_mes"
  OUT=~/Downloads/phase5b_neck_search_bundle.txt

  : > "$OUT"

  append_file() {
    FILE="$1"

    echo "================================================================================" >> "$OUT"

    if [ -f "$PROJECT/$FILE" ]; then
      echo "FILE: $(basename "$FILE")" >> "$OUT"
      echo "PATH: $PROJECT/$FILE" >> "$OUT"
      echo "================================================================================" >> "$OUT"
      cat "$PROJECT/$FILE" >> "$OUT"
    else
      echo "MISSING: $(basename "$FILE")" >> "$OUT"
      echo "PATH: $PROJECT/$FILE" >> "$OUT"
      echo "================================================================================" >> "$OUT"
    fi

    echo >> "$OUT"
  }

  echo "################################################################################" >> "$OUT"
  echo "# src" >> "$OUT"
  echo "################################################################################" >> "$OUT"
  echo >> "$OUT"

  append_file \
    "src/main/java/com/example/guitarmes/neck/NeckService.java"

  append_file \
    "src/main/java/com/example/guitarmes/neck/NeckViewController.java"

  echo "################################################################################" >> "$OUT"
  echo "# test" >> "$OUT"
  echo "################################################################################" >> "$OUT"
  echo >> "$OUT"

  append_file \
    "src/test/java/com/example/guitarmes/e2e/NeckBulkProcessE2E.java"

  append_file \
    "src/test/java/com/example/guitarmes/neck/NeckViewControllerTest.java"

  echo "################################################################################" >> "$OUT"
  echo "# HTML" >> "$OUT"
  echo "################################################################################" >> "$OUT"
  echo >> "$OUT"

  append_file \
    "src/main/resources/templates/neck-list.html"

  append_file \
    "src/main/resources/static/css/style.css"

  echo "Created: $OUT"
}
